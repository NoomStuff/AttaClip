cmake_minimum_required(VERSION 3.24)
project(attaclip_recorder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OBS_SOURCE_DIR "" CACHE PATH "Matching OBS Studio source tree")
set(OBS_IMPORT_LIBRARY "" CACHE FILEPATH "OBS import library")
add_executable(attaclip-recorder recorder.cpp)
target_include_directories(attaclip-recorder PRIVATE vendor)
if(WIN32)
  if(NOT OBS_SOURCE_DIR)
    message(FATAL_ERROR "Provide OBS_SOURCE_DIR and OBS_IMPORT_LIBRARY matching the runtime")
  endif()
  target_include_directories(attaclip-recorder PRIVATE "${OBS_SOURCE_DIR}/libobs" generated)
  target_compile_definitions(attaclip-recorder PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(attaclip-recorder PRIVATE "${OBS_IMPORT_LIBRARY}" user32 psapi ole32)
else()
  find_package(libobs REQUIRED)
  find_package(X11 REQUIRED)
  target_link_libraries(attaclip-recorder PRIVATE OBS::libobs X11::X11)
  set_target_properties(attaclip-recorder PROPERTIES BUILD_RPATH "$ORIGIN/lib" INSTALL_RPATH "$ORIGIN/lib")
endif()
