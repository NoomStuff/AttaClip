cmake_minimum_required(VERSION 3.24)
project(attaclip_recorder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(OBS_SOURCE_DIR "" CACHE PATH "Matching OBS Studio source tree")
set(OBS_IMPORT_LIBRARY "" CACHE FILEPATH "OBS import library")
add_executable(attaclip-recorder recorder.cpp)
target_include_directories(attaclip-recorder PRIVATE vendor)
if(APPLE)
  enable_language(OBJCXX)
  set(OBS_APP "" CACHE PATH "Hash-verified official OBS.app")
  set(SIMDE_SOURCE_DIR "" CACHE PATH "Pinned OBS SIMDe headers")
  if(NOT EXISTS "${OBS_APP}/Contents/Frameworks/libobs.framework/libobs")
    message(FATAL_ERROR "Provide OBS_APP and matching OBS_SOURCE_DIR/SIMDE_SOURCE_DIR")
  endif()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/obsconfig.h" "#pragma once\n#define OBS_RELEASE_CANDIDATE 0\n#define OBS_BETA 0\n")
  target_sources(attaclip-recorder PRIVATE macos/main.mm macos/platform.mm)
  set_source_files_properties(recorder.cpp PROPERTIES COMPILE_DEFINITIONS "main=attaclip_recorder_main")
  target_include_directories(attaclip-recorder PRIVATE "${OBS_SOURCE_DIR}/libobs" "${SIMDE_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(attaclip-recorder PRIVATE "${OBS_APP}/Contents/Frameworks/libobs.framework/libobs" "-framework Cocoa" "-framework CoreGraphics" "-framework AVFoundation" "-framework ScreenCaptureKit" "-framework VideoToolbox")
  target_link_options(attaclip-recorder PRIVATE "-Wl,-sectcreate,__TEXT,__info_plist,${CMAKE_CURRENT_SOURCE_DIR}/macos/Info.plist")
  set_target_properties(attaclip-recorder PROPERTIES BUILD_RPATH "@executable_path/Frameworks;${OBS_APP}/Contents/Frameworks" INSTALL_RPATH "@executable_path/Frameworks" MACOSX_DEPLOYMENT_TARGET "13.0")
elseif(WIN32)
  if(NOT OBS_SOURCE_DIR)
    message(FATAL_ERROR "Provide OBS_SOURCE_DIR and OBS_IMPORT_LIBRARY matching the runtime")
  endif()
  target_include_directories(attaclip-recorder PRIVATE "${OBS_SOURCE_DIR}/libobs" generated)
  target_compile_definitions(attaclip-recorder PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(attaclip-recorder PRIVATE "${OBS_IMPORT_LIBRARY}" user32 psapi ole32)
else()
  find_package(libobs REQUIRED)
  find_package(X11 REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb xcb-composite)
  target_sources(attaclip-recorder PRIVATE x11-compat.cpp)
  target_link_libraries(attaclip-recorder PRIVATE OBS::libobs X11::X11 PkgConfig::XCB)
  set_target_properties(attaclip-recorder PROPERTIES BUILD_RPATH "$ORIGIN/lib" INSTALL_RPATH "$ORIGIN/lib")
endif()
