cmake_minimum_required(VERSION 3.24)
project(attaclip_macos_probe LANGUAGES CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "This probe must compile on macOS")
endif()
set(OBS_APP "" CACHE PATH "Hash-verified official OBS.app")
set(OBS_SOURCE_DIR "" CACHE PATH "Matching pinned OBS source")
if(NOT EXISTS "${OBS_APP}/Contents/Frameworks/libobs.framework/libobs")
  message(FATAL_ERROR "The official libobs framework is missing")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/obsconfig.h" "#pragma once\n#define OBS_RELEASE_CANDIDATE 0\n#define OBS_BETA 0\n")
add_executable(attaclip-macos-probe probe.mm)
target_compile_features(attaclip-macos-probe PRIVATE cxx_std_17)
target_include_directories(attaclip-macos-probe PRIVATE "${OBS_SOURCE_DIR}/libobs" "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../vendor")
target_link_libraries(attaclip-macos-probe PRIVATE "${OBS_APP}/Contents/Frameworks/libobs.framework/libobs" "-framework Cocoa" "-framework CoreGraphics" "-framework AVFoundation")
set_target_properties(attaclip-macos-probe PROPERTIES BUILD_RPATH "${OBS_APP}/Contents/Frameworks")
