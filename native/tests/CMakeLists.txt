cmake_minimum_required(VERSION 3.24)
project(attaclip_native_fixture LANGUAGES CXX)
if(WIN32)
add_executable(attaclip-d3d-fixture d3d-window.cpp)
target_link_libraries(attaclip-d3d-fixture PRIVATE d3d11 dxgi user32 winmm)
target_compile_definitions(attaclip-d3d-fixture PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
elseif(UNIX AND NOT APPLE)
  find_package(X11 REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PULSE_SIMPLE REQUIRED IMPORTED_TARGET libpulse-simple)
  add_executable(attaclip-x11-audio-fixture x11-audio.cpp)
  target_compile_features(attaclip-x11-audio-fixture PRIVATE cxx_std_17)
  target_link_libraries(attaclip-x11-audio-fixture PRIVATE X11::X11 PkgConfig::PULSE_SIMPLE)
else()
  message(FATAL_ERROR "No native fixture for this platform")
endif()
